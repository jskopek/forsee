From the terminal

python -m phonegaze lab  SESSION                      # report: every camera, estimator combination, model
python -m phonegaze lab  SESSION --set lab.eye=left   # any setting, for one run
python -m phonegaze tune SESSION --group crop --group regularisation --save best
python -m phonegaze tune SESSION --auto estimators. --metric loo_sample
python -m phonegaze ab   SESSION --a base --b best    # is the difference larger than the noise?
